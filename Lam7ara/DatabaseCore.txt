PRAGMA foreign_keys = ON;

CREATE TABLE IF NOT EXISTS Clientes (
    ClienteID   INTEGER PRIMARY KEY AUTOINCREMENT,
    Nombre      TEXT    NOT NULL,
    Apellido    TEXT    NOT NULL,
    DNI         TEXT    UNIQUE,
    Telefono    TEXT,
    Email       TEXT,
    FechaAlta   TEXT    NOT NULL DEFAULT (date('now')),
    IsDeleted   INTEGER NOT NULL DEFAULT 0
);


CREATE TABLE IF NOT EXISTS Reparaciones (
    ReparacionID          INTEGER PRIMARY KEY AUTOINCREMENT,
    NombreEquipo          TEXT    NOT NULL,
    NombreDueno           TEXT    NOT NULL,
    ApellidoDueno         TEXT    NOT NULL,
    TelefonoDueno         TEXT,
    DescripcionProblema   TEXT,
    Presupuesto           REAL    NOT NULL DEFAULT 0,
    FechaIngreso          TEXT    NOT NULL DEFAULT (date('now')),
    FechaEstimadaEntrega  TEXT,
    Observaciones         TEXT,
    Estado                TEXT    NOT NULL DEFAULT 'Ingresado'
                          CHECK (Estado IN ('Ingresado', 'En proceso', 'Listo', 'Entregado')),
    IsDeleted             INTEGER NOT NULL DEFAULT 0
);

CREATE INDEX IF NOT EXISTS idx_reparaciones_estado ON Reparaciones (Estado);

CREATE TABLE IF NOT EXISTS Productos (
    ProductoID      INTEGER PRIMARY KEY AUTOINCREMENT,
    Nombre          TEXT    NOT NULL,
    Condicion       TEXT    NOT NULL CHECK (Condicion IN ('Nuevo', 'Usado')),
    Almacenamiento  TEXT,
    Bateria         TEXT,
    Stock           INTEGER NOT NULL DEFAULT 0,
    EmailProveedor  TEXT,
    PrecioCosto     REAL    NOT NULL DEFAULT 0,
    PrecioVenta     REAL    NOT NULL DEFAULT 0,
    Descripcion     TEXT,
    FechaAlta       TEXT    NOT NULL DEFAULT (date('now')),
    IsDeleted       INTEGER NOT NULL DEFAULT 0
);


CREATE TABLE IF NOT EXISTS Cajas (
    CajaID          INTEGER PRIMARY KEY AUTOINCREMENT,
    FechaApertura   TEXT    NOT NULL DEFAULT (datetime('now')),
    MontoApertura   REAL    NOT NULL DEFAULT 0,
    FechaCierre     TEXT,
    MontoCierre     REAL,
    Estado          TEXT    NOT NULL DEFAULT 'Abierta' CHECK (Estado IN ('Abierta', 'Cerrada')),
    Responsable     TEXT
);


CREATE TABLE IF NOT EXISTS Ventas (
    VentaID     INTEGER PRIMARY KEY AUTOINCREMENT,
    ClienteID   INTEGER NOT NULL,
    CajaID      INTEGER NOT NULL,
    Fecha       TEXT    NOT NULL DEFAULT (datetime('now')),
    MedioPago   TEXT    NOT NULL CHECK (MedioPago IN ('Efectivo', 'Transferencia', 'Credito')),
    Cuotas      INTEGER NOT NULL DEFAULT 1,
    Total       REAL    NOT NULL DEFAULT 0,
    Estado      TEXT    NOT NULL DEFAULT 'Completada' CHECK (Estado IN ('Completada', 'Anulada')),
    FOREIGN KEY (ClienteID) REFERENCES Clientes (ClienteID),
    FOREIGN KEY (CajaID)    REFERENCES Cajas    (CajaID)
);


CREATE TABLE IF NOT EXISTS Venta_Productos (
    VentaProductoID INTEGER PRIMARY KEY AUTOINCREMENT,
    VentaID         INTEGER NOT NULL,
    ProductoID      INTEGER NOT NULL,
    Cantidad        INTEGER NOT NULL DEFAULT 1,
    PrecioUnitario  REAL    NOT NULL,
    Subtotal        REAL    NOT NULL,
    FOREIGN KEY (VentaID)    REFERENCES Ventas    (VentaID),
    FOREIGN KEY (ProductoID) REFERENCES Productos (ProductoID)
);


CREATE TABLE IF NOT EXISTS Movimientos (
    MovimientoID    INTEGER PRIMARY KEY AUTOINCREMENT,
    CajaID          INTEGER NOT NULL,
    Tipo            TEXT    NOT NULL CHECK (Tipo IN ('Ingreso', 'Egreso')),
    Monto           REAL    NOT NULL,
    Descripcion     TEXT,
    Responsable     TEXT,
    Fecha           TEXT    NOT NULL DEFAULT (datetime('now')),
    FOREIGN KEY (CajaID) REFERENCES Cajas (CajaID)
);

CREATE INDEX IF NOT EXISTS idx_ventas_cliente   ON Ventas          (ClienteID);
CREATE INDEX IF NOT EXISTS idx_ventas_caja      ON Ventas          (CajaID);
CREATE INDEX IF NOT EXISTS idx_vp_venta         ON Venta_Productos (VentaID);
CREATE INDEX IF NOT EXISTS idx_vp_producto      ON Venta_Productos (ProductoID);
CREATE INDEX IF NOT EXISTS idx_movimientos_caja ON Movimientos     (CajaID);
